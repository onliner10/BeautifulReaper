; Design rules
;  * One layout, adapted automatically. Controls appear when there is room
;    for them or when the track state makes them relevant (e.g. the input
;    selector and monitor button only exist while a track is armed).
;  * Nothing is shown twice. The track panel has a slim meter with no numbers;
;    the mixer has the one full meter with a single scale.
;  * Hit targets are at least 20x20pt. The folder disclosure chevron is the
;    first thing on the row, not a tiny icon in a corner.

#>--------------------------------------------------------------- TRACK PANEL ---
; The track panel is the arrangement header. It holds what you need while
; arranging and recording - nothing that belongs to the mixer. Volume, pan
; and metering live in the mixer only, so there is one place for levels.
;
;   |color| [>] Name ......................... [A] [FX] [M] [S] [o]
;           [ Input 1                        v ]            [mon]    <- armed only

macro drawTcp S F
	clear tcp.*
	set Sv [S S S S S S S S]
	set tcp.size * Sv [260 32 200]

	set step * S 16
	set indent * folderdepth step{0}
	set mx * S 4
	; tree: the whole card (color strip included) moves right one step per folder level
	set panelX + mx{0} indent{0}
	set gap * S 1
	set cardH - h gap{0}
	; content is vertically centred: one row, or two while armed and tall enough
	set armRowH * S 50
	set showRow2 ?recarm cardH{0}<armRowH{0} 0 1 0
	set blockH ?showRow2{0} * S 46 * S 20
	set row1Y * 0.5 - cardH{0} blockH{0}
	set row2Y + row1Y{0} * S 26
	set tinyH * S 18
	set tiny h<tinyH{0} 1 0
	set superH * S 10
	set super h<superH{0} 1 0

	; row 1
	set foldX + panelX{0} * S 8
	set tcp.foldercomp ?tiny{0} [0] folderstate==1 + [foldX{0} row1Y{0}] * Sv [0 0 20 20] [0]
	set nameX + foldX{0} * S 21
	set nameW - - w nameX{0} * S 124
	set tcp.label ?super{0} [0] ?tiny{0} + [nameX{0} 0 nameW{0} cardH{0}] * Sv [0 0 0 0] + [nameX{0} row1Y{0} nameW{0}] * Sv [0 0 0 20]
	set tcp.label.margin * Sv [3 0 4 0]
	set tcp.label.font + [4] [F]
	set tcp.label.color ?track_selected [255 255 255] [$TEXT$]

	set tcp.env ?tiny{0} [0] + [w row1Y{0}] * Sv [-122 0 20 20]
	set tcp.fx ?tiny{0} [0] + [w row1Y{0}] * Sv [-99 0 24 20]
	set tcp.mute ?tiny{0} [0] + [w row1Y{0}] * Sv [-72 0 20 20]
	set tcp.solo ?tiny{0} [0] + [w row1Y{0}] * Sv [-50 0 20 20]
	set tcp.recarm ?tiny{0} [0] + [w row1Y{0}] * Sv [-28 0 20 20]

	; row 2 - recording: input selector + monitoring, only while armed
	set inW - w nameX{0}
	set tcp.recinput !showRow2{0} [0] + [nameX{0} row2Y{0} inW{0}] * Sv [2 0 -58 20]
	set tcp.recinput.font + [2] [F]
	set tcp.recinput.color [$TEXT$]
	set tcp.recinput.margin * Sv [7 0 16 0]
	set tcp.recmon !showRow2{0} [0] + [w row2Y{0}] * Sv [-28 0 20 20]

	; surfaces - declared last so they draw underneath (first declared is on top)
	custom tcp.custom.strip
	set tcp.custom.strip ?super{0} [0] + [panelX{0} 0 0 cardH{0}] * Sv [0 0 3 0]
	set tcp.custom.strip.color ?trackcolor_valid [0 0 0 0 trackcolor_r trackcolor_g trackcolor_b 255] [0 0 0 0 $CONTROL$ 255]
	custom tcp.custom.gap
	set tcp.custom.gap [0 cardH{0} w gap{0}]
	; indent guides, one per ancestor level, centred in that level's step
	custom tcp.custom.guide1
	set tcp.custom.guide1 folderdepth<1 [0] + [mx{0} 0 0 h] * Sv [8 0 1 0]
	set tcp.custom.guide1.color [0 0 0 0 $GUIDE$ 255]
	custom tcp.custom.guide2
	set tcp.custom.guide2 folderdepth<2 [0] + [mx{0} 0 0 h] * Sv [24 0 1 0]
	set tcp.custom.guide2.color [0 0 0 0 $GUIDE$ 255]
	custom tcp.custom.guide3
	set tcp.custom.guide3 folderdepth<3 [0] + [mx{0} 0 0 h] * Sv [40 0 1 0]
	set tcp.custom.guide3.color [0 0 0 0 $GUIDE$ 255]
	custom tcp.custom.guide4
	set tcp.custom.guide4 folderdepth<4 [0] + [mx{0} 0 0 h] * Sv [56 0 1 0]
	set tcp.custom.guide4.color [0 0 0 0 $GUIDE$ 255]
	custom tcp.custom.gutter
	set tcp.custom.gutter [0 0 panelX{0} h]
	set tcp.custom.gutter.color [0 0 0 0 $BG_DEEP$ 255]
	set tcp.custom.gap.color [0 0 0 0 $BG_DEEP$ 255]
	custom tcp.custom.bg
	set tcp.custom.bg [0 0 w cardH{0}]
	set tcp.custom.bg.color ?super{0} [0 0 0 0 $BG_DEEP$ 255] ?track_selected [0 0 0 0 $PANEL_SEL$ 255] [0 0 0 0 $PANEL$ 255]
	front tcp.foldercomp
endmacro

drawTcp 1 0
Layout "A"
	drawTcp 1 0
endLayout
Layout "200%_A" "200"
	drawTcp 2 10
endLayout

macro drawMcp S F
	clear mcp.*
	set Sv [S S S S S S S S]
	set mcp.size * Sv [80 400]

	; name (with the folder chevron for folder parents)
	set mcp.foldercomp folderstate==1 * Sv [1 5 20 20] [0]
	set mcp.folder folderstate==1 * Sv [1 5 20 20] [0]
	set labelX folderstate==1 * S 18 0
	set labelW - w labelX{0}
	set mcp.label + [labelX{0} 0 labelW{0}] * Sv [0 5 0 20]
	set mcp.label.margin * Sv [3 0 3 0 0.5]
	set mcp.label.font + [3] [F]
	set mcp.label.color ?track_selected [255 255 255] [$TEXT$]

	; vertical budget: name | inserts+sends (flexible) | pan | fader (min 130) | buttons
	set top * S 29
	set btnBlock * S 56
	set panBlock * S 40
	set avail - - - h top{0} btnBlock{0} panBlock{0}
	set faderH * 0.6 avail{0}
	set minF * S 110
	set faderH faderH{0}<minF{0} minF{0} .
	set extH - avail{0} faderH{0}
	set extH extH{0}<0 0 .

	; inserts / sends
	set mcp.extmixer.mode [1]
	set mcp.extmixer.position + [0 top{0} w extH{0}] * Sv [3 0 -6 0]
	set mcp.fxlist.font + [2] [F]
	set mcp.fxlist.margin * Sv [5 0 4 0 0 0 0 0]
	set mcp.sendlist.font + [2] [F]
	set mcp.sendlist.margin * Sv [5 0 4 0 0 0 0 0]
	set mcp.fxparm.font + [1] [F]

	; pan
	set panY + top{0} extH{0}
	set halfW * 0.5 w
	set mcp.pan + [halfW{0} panY{0}] * Sv [-13 7 26 26]
	set mcp.pan.fadermode [1]

	; fader + the one real meter
	set faderY + panY{0} panBlock{0}
	set mcp.volume.label + [0 faderY{0}] * Sv [0 0 42 14]
	set mcp.volume.label.margin [0 0 0 0 0.5]
	set mcp.volume.label.font + [2] [F]
	set mcp.volume.label.color [$TEXT$]
	set mcp.volume + [0 faderY{0} 0 faderH{0}] * Sv [7 18 26 -22]
	set mcp.volume.fadermode [0]
	set mcp.meter + [0 faderY{0} 0 faderH{0}] * Sv [42 2 32 -6]
	set mcp.meter.vu.div * Sv [16]
	set mcp.meter.readout.color [$TEXT_2$ 255 $METER_CLIP$ 255]
	set mcp.meter.scale.color.lit.bottom [$TEXT_3$ 255 0 0 0 0]
	set mcp.meter.scale.color.lit.top [$TEXT_3$ 255 0 0 0 0]
	set mcp.meter.scale.color.unlit.bottom [$TEXT_3$ 255 0 0 0 0]
	set mcp.meter.scale.color.unlit.top [$TEXT_3$ 255 0 0 0 0]

	; buttons: mute, solo, arm + routing
	set btnY - h * S 56
	set mcp.mute + [0 btnY{0}] * Sv [6 0 20 20]
	set mcp.solo + [0 btnY{0}] * Sv [30 0 20 20]
	set mcp.recarm + [0 btnY{0}] * Sv [54 0 20 20]
	set mcp.io + [0 btnY{0}] * Sv [30 24 20 20]

	custom mcp.custom.colorBar
	set mcp.custom.colorBar + [0 0 w] * Sv [0 0 0 3]
	set mcp.custom.colorBar.color ?trackcolor_valid [0 0 0 0 trackcolor_r trackcolor_g trackcolor_b 255] [0 0 0 0 $CONTROL$ 255]
	custom mcp.custom.div
	set mcp.custom.div + [w 0 1 h] [-1]
	set mcp.custom.div.color [0 0 0 0 $DIVIDER$ 255]
	custom mcp.custom.bg
	set mcp.custom.bg [0 0 w h]
	set mcp.custom.bg.color ?track_selected [0 0 0 0 $PANEL_SEL$ 255] [0 0 0 0 $PANEL$ 255]
	front mcp.foldercomp
endmacro
