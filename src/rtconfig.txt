; Design rules
;  * One layout, adapted automatically. Controls appear when there is room
;    for them or when the track state makes them relevant (e.g. the input
;    selector and monitor button only exist while a track is armed).
;  * Nothing is shown twice. The track panel has a slim meter with no numbers;
;    the mixer has the one full meter with a single scale.
;  * Hit targets are at least 20x20pt. The folder disclosure chevron is the
;    first thing on the row, not a tiny icon in a corner.

#>--------------------------------------------------------------- TRACK PANEL ---
; The track panel is the complete channel: everything a producer needs while
; arranging, recording and balancing lives here, so the mixer can stay closed
; and the screen height goes to tracks. Wide screens have room to spare
; horizontally; the panel spends it on one row of controls.
;
;   |color| [>] Name ..... [==meter==|==] -2.1 (pan) [io] [A] [FX] [M] [S] [o]
;           [ Input 1                                   v ]           [mon]   <- armed only
;
; When the panel is narrow the fader goes first, then the name shortens.

macro drawTcp S F
	clear tcp.*
	set Sv [S S S S S S S S]
	set tcp.size * Sv [440 32 240]

	set step * S 16
	set indent * folderdepth step{0}
	set mx * S 4
	; tree: the whole card (color strip included) moves right one step per folder level
	set panelX + mx{0} indent{0}
	set gap * S 1
	set cardH - h gap{0}
	set tinyH * S 18
	set tiny h<tinyH{0} 1 0
	set superH * S 10
	set super h<superH{0} 1 0

	set foldX + panelX{0} * S 8
	set nameX + foldX{0} * S 21
	set minNameR + nameX{0} * S 72

	; ---- responsive rows ------------------------------------------------------
	; Row 1 always holds: [>] Name ... [I/O] [A] [FX] [M] [S] [o]
	; The level block (sends, fader/meter, dB, pan) sits on row 1 when the panel
	; is wide enough, otherwise it wraps to its own row if the track is tall
	; enough. Armed tracks add an input row; if only two rows fit, recording
	; wins and the wrapped level row hides.
	set rowStep * S 26
	; the level block has one fixed size everywhere (fader 120pt, meter 110pt) so the
	; LED segments are drawn pixel-exact; if it does not fit on row 1 it wraps
	set volX1 - w * S 302
	set minVolX1 + minNameR{0} * S 46
	set wrap ?tiny{0} 0 volX1{0}<minVolX1{0} 1 0
	set h2 * S 46
	set h3 * S 72
	set can2 cardH{0}<h2{0} 0 1
	set can3 cardH{0}<h3{0} 0 1
	set armed ?recarm 1 0
	set levelRow ?tiny{0} 0 ?wrap{0} ?armed{0} ?can3{0} 2 0 ?can2{0} 2 0 1
	set inputRow ?armed{0} levelRow{0}==2 3 ?can2{0} 2 0 0
	set rows inputRow{0}>levelRow{0} inputRow{0} levelRow{0}
	set rows rows{0}<1 1 .
	set blockH - * rows{0} rowStep{0} * S 6
	set row1Y * 0.5 - cardH{0} blockH{0}
	set levelY + row1Y{0} * - levelRow{0} 1 rowStep{0}
	set inputY + row1Y{0} * - inputRow{0} 1 rowStep{0}

	; ---- row 1 ------------------------------------------------------------------
	set tcp.foldercomp ?tiny{0} [0] folderstate==1 + [foldX{0} row1Y{0}] * Sv [0 0 20 20] [0]

	; level block on row 1: fader right edge fixed at w-226, left edge at w-346
	; unless the name needs room (then it shrinks)
	; row 1 reads left to right: name | sends | -2.1 dB | meter+fader | pan
	set sendX1 - volX1{0} * S 226
	; level block on its own row: sends, then a wide fader, dB and pan at the right
	set sendW * S 172
	set hasSends !send_cnt 0 1
	set volX2 - w * S 156

	set volX levelRow{0}==2 volX2{0} volX1{0}
	set volW * S 120
	set showVol levelRow{0}==0 0 1
	set sendX levelRow{0}==2 - volX{0} * S 226 sendX1{0}
	set showSends !hasSends{0} 0 !showVol{0} 0 levelRow{0}==2 sendX{0}<nameX{0} 0 1 sendX{0}<minNameR{0} 0 1

	set nameR levelRow{0}==2 - w * S 156 levelRow{0}==1 ?showSends{0} sendX{0} ?showVol{0} - volX{0} * S 46 - w * S 182 - w * S 182
	set nameW - - nameR{0} nameX{0} * S 8
	set tcp.label ?super{0} [0] ?tiny{0} + [nameX{0} 0 nameW{0} cardH{0}] * Sv [0 0 0 0] + [nameX{0} row1Y{0} nameW{0}] * Sv [0 0 0 20]
	set tcp.label.margin * Sv [3 0 4 0]
	set tcp.label.font + [2] [F]
	set tcp.label.color ?track_selected [255 255 255] [$TEXT$]

	; level: the meter is the fader's track
	set tcp.volume !showVol{0} [0] + [volX{0} levelY{0} volW{0}] * Sv [0 0 0 20]
	set tcp.volume.fadermode [0]
	; meter: 12pt tall, two 5pt channels; the fader is a marker whose tabs show above and below it
	set tcp.meter !showVol{0} [0] + [volX{0} levelY{0}] * Sv [5 4 110 12]
	set tcp.meter.vu.div * Sv [2]
	set tcp.meter.readout.color [0 0 0 0 0 0 0 0]
	set tcp.meter.scale.color.lit.bottom [0 0 0 0 0 0 0 0]
	set tcp.meter.scale.color.lit.top [0 0 0 0 0 0 0 0]
	set tcp.meter.scale.color.unlit.bottom [0 0 0 0 0 0 0 0]
	set tcp.meter.scale.color.unlit.top [0 0 0 0 0 0 0 0]
	set dbX - volX{0} * S 46
	set tcp.volume.label !showVol{0} [0] + [dbX{0} levelY{0}] * Sv [0 0 42 20]
	set tcp.volume.label.margin [0 0 0 0 1]
	set tcp.volume.label.font + [1] [F]
	set tcp.volume.label.color [$TEXT_2$]
	set panX levelRow{0}==2 * S -28 * S -176
	set panY levelRow{0}==2 levelY{0} row1Y{0}
	set tcp.pan ?tiny{0} [0] + [w panY{0}] + [panX{0}] * Sv [0 0 20 20]
	set tcp.pan.fadermode [1]

	; sends: compact name + level slots, only when the track has sends
	set tcp.sendlist !showSends{0} [0] + [sendX{0} levelY{0} sendW{0}] * Sv [0 0 0 20]
	set tcp.sendlist.font + + [1] [F] * Sv [0 20 84 172]
	set tcp.sendlist.margin * Sv [2 0 2 0 0 0 0 0]

	set tcp.io ?tiny{0} [0] + [w row1Y{0}] * Sv [-150 0 24 20]
	set tcp.env ?tiny{0} [0] + [w row1Y{0}] * Sv [-122 0 20 20]
	set tcp.fx ?tiny{0} [0] + [w row1Y{0}] * Sv [-99 0 24 20]
	set tcp.mute ?tiny{0} [0] + [w row1Y{0}] * Sv [-72 0 20 20]
	set tcp.solo ?tiny{0} [0] + [w row1Y{0}] * Sv [-50 0 20 20]
	set tcp.recarm ?tiny{0} [0] + [w row1Y{0}] * Sv [-28 0 20 20]

	; ---- input row (armed only): input selector + monitoring --------------------
	set inW - w nameX{0}
	set tcp.recinput inputRow{0}==0 [0] + [nameX{0} inputY{0} inW{0}] * Sv [2 0 -58 20]
	set tcp.recinput.font + [1] [F]
	set tcp.recinput.color [$TEXT$]
	set tcp.recinput.margin * Sv [7 0 16 0]
	set tcp.recmon inputRow{0}==0 [0] + [w inputY{0}] * Sv [-28 0 20 20]

	; surfaces - declared last so they draw underneath (first declared is on top)
	custom tcp.custom.strip
	set tcp.custom.strip ?super{0} [0] + [panelX{0} 0 0 cardH{0}] * Sv [0 0 3 0]
	set tcp.custom.strip.color ?trackcolor_valid [0 0 0 0 trackcolor_r trackcolor_g trackcolor_b 255] [0 0 0 0 $CONTROL$ 255]
	custom tcp.custom.gap
	set tcp.custom.gap [0 cardH{0} w gap{0}]
	; indent guides, one per ancestor level, centred in that level's step
	custom tcp.custom.guide1
	set tcp.custom.guide1 folderdepth<1 [0] + [mx{0} 0 0 h] * Sv [8 0 1 0]
	set tcp.custom.guide1.color [0 0 0 0 $GUIDE$ 255]
	custom tcp.custom.guide2
	set tcp.custom.guide2 folderdepth<2 [0] + [mx{0} 0 0 h] * Sv [24 0 1 0]
	set tcp.custom.guide2.color [0 0 0 0 $GUIDE$ 255]
	custom tcp.custom.guide3
	set tcp.custom.guide3 folderdepth<3 [0] + [mx{0} 0 0 h] * Sv [40 0 1 0]
	set tcp.custom.guide3.color [0 0 0 0 $GUIDE$ 255]
	custom tcp.custom.guide4
	set tcp.custom.guide4 folderdepth<4 [0] + [mx{0} 0 0 h] * Sv [56 0 1 0]
	set tcp.custom.guide4.color [0 0 0 0 $GUIDE$ 255]
	custom tcp.custom.gutter
	set tcp.custom.gutter [0 0 panelX{0} h]
	set tcp.custom.gutter.color [0 0 0 0 $BG_DEEP$ 255]
	set tcp.custom.gap.color [0 0 0 0 $BG_DEEP$ 255]
	custom tcp.custom.bg
	set tcp.custom.bg [0 0 w cardH{0}]
	set tcp.custom.bg.color ?super{0} [0 0 0 0 $BG_DEEP$ 255] ?track_selected [0 0 0 0 $PANEL_SEL$ 255] [0 0 0 0 $PANEL$ 255]
	; front: the first element listed ends up on top
	front tcp.volume tcp.meter tcp.foldercomp
endmacro

$LAYOUTS drawTcp$
